async function recoverLostData() { let localCust = JSON.parse(localStorage.getItem('ncn_customers')) || {}; let recordsToInsert = []; for (let phone in localCust) { let cust = localCust[phone]; if (cust.history && cust.history.length > 0) { cust.history.forEach(h => { recordsToInsert.push({ id: 'rec_recover_' + Date.now().toString() + Math.floor(Math.random() * 1000), phone: phone, name: cust.name, branch_id: 'unknown', branch_name: h.branchName, reward_name: h.rewardName, time: h.time }); }); } } if (recordsToInsert.length > 0) { console.log("กำลังกู้ข้อมูลขึ้นระบบจำนวน: " + recordsToInsert.length + " รายการ..."); const { error } = await supabaseClient.from('draw_records').insert(recordsToInsert); if (error) { console.error("เกิดข้อผิดพลาด:", error); alert("กู้ข้อมูลไม่สำเร็จ กรุณาดู Error ใน Console"); } else { console.log("✅ กู้ข้อมูลสำเร็จ!"); alert("กู้ข้อมูลขึ้นระบบสำเร็จแล้ว! ข้อมูลจะไปโชว์ในเครื่องอื่นด้วยแล้วครับ"); } } else { alert("ไม่พบข้อมูลตกค้างในเครื่องนี้ครับ"); } } recoverLostData();
